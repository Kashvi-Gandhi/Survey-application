-- -------------------------------------------------------------
-- 1. EXTENSIONS & ENUMS
-- -------------------------------------------------------------
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- Enum for user roles
CREATE TYPE user_role_enum AS ENUM ('admin', 'surveyor');

-- Enum for question types
CREATE TYPE question_type_enum AS ENUM (
  'single_select',
  'multi_select',
  'true_false',
  'one_line',
  'rating'
);

-- -------------------------------------------------------------
-- 2. TABLES
-- -------------------------------------------------------------

-- A. ROLE MASTER TABLE
CREATE TABLE public.role_master (
    id SERIAL PRIMARY KEY,
    role_name user_role_enum UNIQUE NOT NULL,
    description TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

-- B. USER PROFILES TABLE (Linked to Supabase auth.users)
CREATE TABLE public.profiles (
    id UUID PRIMARY KEY REFERENCES auth.users(id) ON DELETE CASCADE,
    email VARCHAR(255) UNIQUE NOT NULL,
    full_name VARCHAR(255) NOT NULL,
    role_id INT NOT NULL REFERENCES public.role_master(id) ON DELETE RESTRICT,
    is_active BOOLEAN DEFAULT TRUE NOT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

-- C. QUESTION BANKS TABLE
CREATE TABLE public.question_banks (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    title VARCHAR(255) NOT NULL,
    description TEXT,
    created_by UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

-- D. SURVEYS / QUIZZES TABLE
CREATE TABLE public.surveys (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    title VARCHAR(255) NOT NULL,
    description TEXT,
    time_limit_minutes INT DEFAULT 0, -- 0 means untimed
    is_published BOOLEAN DEFAULT FALSE NOT NULL,
    expires_at TIMESTAMPTZ,
    created_by UUID NOT NULL REFERENCES public.profiles(id) ON DELETE CASCADE,
    created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

-- E. QUESTIONS TABLE (Embedded Options & Correct Answers via JSONB)
CREATE TABLE public.questions (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    bank_id UUID REFERENCES public.question_banks(id) ON DELETE SET NULL,
    survey_id UUID REFERENCES public.surveys(id) ON DELETE CASCADE,
    question_text TEXT NOT NULL,
    question_type question_type_enum NOT NULL,
    order_index INT DEFAULT 1 NOT NULL,
    points INT DEFAULT 1 NOT NULL,
    is_required BOOLEAN DEFAULT TRUE NOT NULL,
    
    /* 
      JSONB Schema Examples for `options`:
      - single_select / multi_select: ["Option A", "Option B", "Option C"]
      - true_false: ["True", "False"]
      - rating: {"min": 1, "max": 5, "min_label": "Poor", "max_label": "Excellent"}
      - one_line: null
    */
    options JSONB DEFAULT NULL,
    
    /* 
      JSONB Schema Examples for `correct_answer`:
      - single_select / true_false: "Option A"
      - multi_select: ["Option A", "Option C"]
      - rating / one_line: "5" or "Sample exact answer" (or null if purely survey)
    */
    correct_answer JSONB DEFAULT NULL,
    
    created_at TIMESTAMPTZ DEFAULT NOW() NOT NULL,
    updated_at TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

-- F. RESPONSES / SUBMISSIONS TABLE
CREATE TABLE public.responses (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    survey_id UUID NOT NULL REFERENCES public.surveys(id) ON DELETE CASCADE,
    question_id UUID NOT NULL REFERENCES public.questions(id) ON DELETE CASCADE,
    respondent_id UUID REFERENCES public.profiles(id) ON DELETE SET NULL, -- Nullable for anonymous responses
    
    /* 
      JSONB Schema Examples for `user_answer`:
      - single_select / true_false / one_line: "Option A" or "User text answer"
      - multi_select: ["Option A", "Option B"]
      - rating: 4
    */
    user_answer JSONB NOT NULL,
    is_correct BOOLEAN DEFAULT NULL, -- Pre-computed or evaluated post-submission
    score_earned INT DEFAULT 0 NOT NULL,
    submitted_at TIMESTAMPTZ DEFAULT NOW() NOT NULL
);

-- -------------------------------------------------------------
-- 4. AUTOMATIC USER SYNC (Supabase Auth -> Public Profiles)
-- -------------------------------------------------------------
CREATE OR REPLACE FUNCTION public.handle_new_user()
RETURNS TRIGGER AS $$
DECLARE
  default_role_id INT;
BEGIN
  SELECT id INTO default_role_id FROM public.role_master WHERE role_name = 'surveyor';
  
  INSERT INTO public.profiles (id, email, full_name, role_id)
  VALUES (
    NEW.id,
    NEW.email,
    COALESCE(NEW.raw_user_meta_data->>'full_name', 'New User'),
    default_role_id
  );
  RETURN NEW;
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;

CREATE OR REPLACE TRIGGER on_auth_user_created
  AFTER INSERT ON auth.users
  FOR EACH ROW EXECUTE FUNCTION public.handle_new_user();

-- -------------------------------------------------------------
-- 5. EXAMPLE STORED PROCEDURE (PL/pgSQL Function)
-- -------------------------------------------------------------
-- Procedure: Submit a single response and auto-calculate correctness
CREATE OR REPLACE FUNCTION public.submit_question_response(
    p_survey_id UUID,
    p_question_id UUID,
    p_respondent_id UUID,
    p_user_answer JSONB
)
RETURNS JSONB AS $$
DECLARE
    v_correct_answer JSONB;
    v_points INT;
    v_is_correct BOOLEAN := FALSE;
    v_score_earned INT := 0;
    v_response_id UUID;
BEGIN
    -- Fetch question metadata
    SELECT correct_answer, points 
    INTO v_correct_answer, v_points
    FROM public.questions 
    WHERE id = p_question_id;

    -- Evaluate correctness if a correct answer is defined
    IF v_correct_answer IS NOT NULL THEN
        IF v_correct_answer = p_user_answer THEN
            v_is_correct := TRUE;
            v_score_earned := v_points;
        END IF;
    END IF;

    -- Insert response record
    INSERT INTO public.responses (
        survey_id,
        question_id,
        respondent_id,
        user_answer,
        is_correct,
        score_earned
    ) VALUES (
        p_survey_id,
        p_question_id,
        p_respondent_id,
        p_user_answer,
        v_is_correct,
        v_score_earned
    )
    RETURNING id INTO v_response_id;

    RETURN jsonb_build_object(
        'success', TRUE,
        'response_id', v_response_id,
        'is_correct', v_is_correct,
        'score_earned', v_score_earned
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
